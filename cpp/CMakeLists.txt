cmake_minimum_required(VERSION 3.24)
project(skeleton VERSION 0.1.0 LANGUAGES CXX)

find_package(nlohmann_json 3.12 REQUIRED)
find_package(OpenSSL CONFIG REQUIRED)
include(CTest)
if(BUILD_TESTING)
    find_package(GTest REQUIRED)
    include(GoogleTest)
    add_library(skeleton_test_support INTERFACE)
    target_include_directories(skeleton_test_support INTERFACE tests/support)
    target_compile_definitions(skeleton_test_support INTERFACE
        SITE_FIXTURE_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/site.json")
    target_link_libraries(skeleton_test_support INTERFACE skeleton_input GTest::gtest_main)
endif()

add_subdirectory(libs/contracts)
add_subdirectory(libs/installation)
add_subdirectory(libs/vm)
add_subdirectory(libs/placement)
add_subdirectory(libs/cleanup)
add_subdirectory(libs/backup)
add_subdirectory(libs/networks)
add_subdirectory(libs/topology)
add_subdirectory(libs/input)
add_subdirectory(libs/planning)
add_subdirectory(libs/tofu)
add_subdirectory(libs/ansible)
add_subdirectory(libs/k0s)
add_subdirectory(libs/graphviz)
add_subdirectory(libs/report)
add_subdirectory(libs/output)
add_subdirectory(libs/pipeline)

add_executable(config-generator app/main.cpp)
target_link_libraries(config-generator PRIVATE skeleton_pipeline)
install(TARGETS config-generator RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
